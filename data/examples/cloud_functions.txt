  @override
  _MyAppState createState() => _MyAppState();

class _MyAppState extends State {
  List fruit = [];
  List streamResult = [];

  @override
  void initState() {
    super.initState();
    streamFunction();
  }

  void streamFunction() {
    fruit.clear();
    FirebaseFunctions.instance
        .httpsCallable('testStreamResponse')
        .stream>()
        .listen(
      (data) {
        switch (data) {
          case Chunk>(:final partialData):
            setState(() {
              &#47;&#47; adds individual stream values to list
              fruit.add(partialData);
            });
          case Result>(:final result):
            setState(() {
              &#47;&#47; stores complete stream result
              streamResult = List.from(result.data);
            });
        }
      },
      onError: (e) {
        debugPrint('Error: $e');
      },
    );
  }
