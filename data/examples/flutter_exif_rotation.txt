import 'dart:async';
import 'dart:io';

import 'package:flutter&#47;material.dart';
import 'package:flutter_exif_rotation&#47;flutter_exif_rotation.dart';
import 'package:image_picker&#47;image_picker.dart';

void main() => runApp(MyApp());

&#47;&#47;&#47; The stateful widget
class MyApp extends StatefulWidget {
  @override
  _MyAppState createState() => _MyAppState();
}

&#47;&#47;&#47; The class with the scaffold
class _MyAppState extends State {
  File? _image;
  final picker = ImagePicker();

  Future getImage() async {
    final image = await picker.getImage(source: ImageSource.gallery);
    if (image != null) {
      File rotatedImage =
          await FlutterExifRotation.rotateImage(path: image.path);

      setState(() {
        _image = rotatedImage;
      });
    }
  }

  Future getImageAndSave() async {
    final image = await picker.getImage(source: ImageSource.gallery);
    if (image != null) {
      File rotatedImage =
          await FlutterExifRotation.rotateAndSaveImage(path: image.path);

      setState(() {
        _image = rotatedImage;
      });
    }
  }

  @override
  void initState() {
    super.initState();
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          title: const Text('Exif flutter rotation image example app'),
        ),
        body: new Center(
          child: _image == null
              ? new Text('No image selected.')
              : new Image.file(_image!),
        ),
        persistentFooterButtons: [
          new FloatingActionButton(
            onPressed: getImageAndSave,
            tooltip: 'Pick Image and save',
            child: new Icon(Icons.save),
          ),
          new FloatingActionButton(
            onPressed: getImage,
            tooltip: 'Pick Image without saving',
            child: new Icon(Icons.add),
          ),
        ],
      ),
    );
  }
}
