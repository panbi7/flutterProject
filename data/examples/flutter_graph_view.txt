    runApp(const Showroom());
  }, (error, stackTrace) {
    &#47;&#47; ... 错误处理
    print(stackTrace);
  });
}

class Showroom extends StatefulWidget {
  const Showroom({super.key});

  @override
  State createState() => _ShowroomState();
}

class _ShowroomState extends State
    with SingleTickerProviderStateMixin {
  late TabController mainTabController;

  final List tabNames = [
    'ErGraphDemo',
    'ObjectGraphDemo',
    'DecoratorDemo',
    'PersistenceDemo',
    'ForceDirected',
    'ExecutionPlan',
    'CircleLayout',
    'RandomLayout',
    "SelfLoop",
    "VertexFontStyleDemo",
  ];
  final List tabs = [
    ErGraphDemo(),
    const ObjectGraphDemo(),
    const DecoratorDemo(),
    const PersistenceDemo(),
    ForceDirectedDemo(),
    ExecutionPlanDemo(),
    CircleLayoutDemo(),
    RandomAlgorithmDemo(),
    const SelfLoopDemo(),
    const VertexFontStyleDemo(),
  ];

  @override
  void initState() {
    super.initState();
    mainTabController =
        TabController(initialIndex: 0, length: tabs.length, vsync: this);
  }

  Uri repo = Uri.parse('https:&#47;&#47;github.com&#47;graph-cn&#47;flutter_graph_view');

  List buttons = [
    {
      'label': 'Repo',
      'url': 'https:&#47;&#47;github.com&#47;graph-cn&#47;flutter_graph_view',
      'icon': Icons.link,
    },
    {
      'label': 'Demos',
      'url':
          'https:&#47;&#47;github.com&#47;graph-cn&#47;flutter_graph_view&#47;tree&#47;main&#47;example&#47;lib&#47;demos',
      'icon': Icons.list,
    }
  ];

  _launchUrl(String url) {
    launchUrl(Uri.parse(url));
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData.dark(useMaterial3: true),
      home: Scaffold(
        body: Column(
          children: [
            Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: buttons.map((button) {
                  return ElevatedButton.icon(
                    icon: Icon(button['icon']),
                    label: Text(button['label']),
                    onPressed: () => _launchUrl(button['url']),
                  );
                }).toList()),
            Expanded(
                child: TabBarView(
              controller: mainTabController,
              children: tabs,
            ))
          ],
        ),
        bottomNavigationBar: TabBar(
          controller: mainTabController,
          tabs: tabNames
              .map((tabName) => Tab(
                    child: Tooltip(message: tabName, child: Text(tabName)),
                  ))
              .toList(),
        ),
      ),
    );
  }
}
