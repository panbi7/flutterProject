class LinearGaugeExample extends StatefulWidget {
  const LinearGaugeExample({Key? key}) : super(key: key);

  @override
  State createState() => _LinearGaugeExampleState();
}

class _LinearGaugeExampleState extends State {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(
        child: LinearGauge(
          gaugeOrientation: GaugeOrientation.horizontal,
          enableGaugeAnimation: true,
          rulers: RulerStyle(
            rulerPosition: RulerPosition.bottom,
          ),
          pointers: const [
            Pointer(
              value: 50,
              shape: PointerShape.circle,
            ),
          ],
        ),
      ),
    );
  }
}

class RadialGaugeExample extends StatefulWidget {
  const RadialGaugeExample({super.key});

  @override
  State createState() => _RadialGaugeExampleState();
}

class _RadialGaugeExampleState extends State {
  @override
  Widget build(BuildContext context) {
    return const Scaffold(
      backgroundColor: Colors.white,
      body: RadialGauge(
        track: RadialTrack(
          start: 0,
          end: 100,
        ),
        needlePointer: [
          NeedlePointer(
            value: 30,
          ),
        ],
      ),
    );
  }
}
