class MyApp extends StatefulWidget {
  const MyApp({Key? key}) : super(key: key);

  @override
  _MyAppState createState() => _MyAppState();
}

class _PiningSslData {
  String serverURL = '';
  Map headerHttp = {};
  String allowedSHAFingerprint = '';
  int timeout = 0;
  SHA? sha;
}

class _MyAppState extends State {
  final GlobalKey _formKey = GlobalKey();
  final _PiningSslData _data = _PiningSslData();
  final _messengerKey = GlobalKey();

  @override
  initState() {
    super.initState();
  }

  &#47;&#47; Platform messages are asynchronous, so we initialize in an async method.
  check(
    String url,
    String fingerprint,
    SHA sha,
    Map headerHttp,
    int timeout,
  ) async {
    List allowedShA1FingerprintList = [];
    allowedShA1FingerprintList.add(fingerprint);

    try {
      &#47;&#47; Platform messages may fail, so we use a try&#47;catch PlatformException.
      String checkMsg = await HttpCertificatePinning.check(
          serverURL: url,
          headerHttp: headerHttp,
          sha: sha,
          allowedSHAFingerprints: allowedShA1FingerprintList,
          timeout: timeout);

      &#47;&#47; If the widget was removed from the tree while the asynchronous platform
      &#47;&#47; message was in flight, we want to discard the reply rather than calling
      &#47;&#47; setState to update our non-existent appearance.
      if (!mounted) return;

      _messengerKey.currentState?.showSnackBar(
        SnackBar(
          content: Text(checkMsg),
          duration: const Duration(seconds: 1),
          backgroundColor: Colors.green,
        ),
      );
    } catch (e) {
      _messengerKey.currentState?.showSnackBar(
        SnackBar(
          content: Text(e.toString()),
          duration: const Duration(seconds: 1),
          backgroundColor: Colors.red,
        ),
      );
    }
  }

  void submit() {
    &#47;&#47; First validate form.
    if (_formKey.currentState?.validate() == true) {
      _formKey.currentState?.save(); &#47;&#47; Save our form now.

      check(
        _data.serverURL,
        _data.allowedSHAFingerprint,
        _data.sha ?? SHA.SHA256,
        _data.headerHttp,
        _data.timeout,
      );
    }
  }
