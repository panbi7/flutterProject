  String notificationTitle = 'MyApp';

  &#47;&#47; Attempt to extract the "message" property from the payload: {"message":"Hello World!"}
  String notificationText = data['message'] ?? 'Hello World!';

  &#47;&#47; Android: Displays a system notification
