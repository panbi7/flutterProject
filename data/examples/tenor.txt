void printTenorResponse(TenorResponse? res) {
  res?.results.forEach((tenorResult) {
    var title = tenorResult.title;
    var media = tenorResult.media;
    print('$title: gif   ${i++}   : ${media?.gif?.previewUrl?.toString()}');
  });
}
