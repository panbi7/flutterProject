  runApp(MyApp());

class MyApp extends StatelessWidget {

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter UniversalImage',
      theme: ThemeData(
        primarySwatch: Colors.blue,
      ),
      home: MyHomePage(title: 'Demo UniversalImage'),
    );
  }

class MyHomePage extends StatefulWidget {
  MyHomePage({Key? key, required this.title}) : super(key: key);

  final String title;

  @override
  _MyHomePageState createState() => _MyHomePageState();
}

class _MyHomePageState extends State {
  Future _loadBase64Image() async {
    var bytes =
        (await rootBundle.load('assets&#47;sample.png')).buffer.asUint8List();
    return bytes.base64Image;
  }

  Future _loadBytesImage() async {
    var bytes =
        (await rootBundle.load('assets&#47;sample.png')).buffer.asUint8List();
    return bytes;
  }

  @override
  Widget build(BuildContext context) {
    const space = 10.0;
    return Scaffold(
        appBar: AppBar(
          title: Text(widget.title),
        ),
        body: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              &#47;&#47; icon provider
              UniversalImage(
                Icons.schedule,
                color: Colors.red,
              ),
              SizedBox(height: space),
              &#47;&#47; svg provider
              UniversalImage(
                'assets&#47;sample.svg',
                width: 30,
                height: 30,
                fit: BoxFit.cover,
              ),
              SizedBox(height: space),
              &#47;&#47; normal provider
              UniversalImage(
                'assets&#47;sample.png',
                width: 100,
                height: 100,
                fit: BoxFit.cover,
                placeholder: Container(),
              ),
              SizedBox(height: space),
              &#47;&#47; base64 provider
              FutureBuilder(
                future: _loadBase64Image(),
                builder: (context, snapshot) {
                  if (snapshot.hasData) {
                    return UniversalImage(
                      snapshot.data!,
                      width: 100,
                      height: 100,
                      fit: BoxFit.cover,
                    );
                  }

                  return Container();
                },
              ),
              SizedBox(height: space),
              &#47;&#47; memory bytes provider
              FutureBuilder(
                future: _loadBytesImage(),
                builder: (context, snapshot) {
                  if (snapshot.hasData) {
                    return UniversalImage(
                      snapshot.data!,
                      width: 100,
                      height: 100,
                      fit: BoxFit.cover,
                    );
                  }

                  return Container();
                },
              )
            ],
          ),
        ));
  }
}
