class _MyAppState extends State {
  List accessPoints = [];
  StreamSubscription>? subscription;
  bool shouldCheckCan = true;

  bool get isStreaming => subscription != null;

  Future _startScan(BuildContext context) async {
    &#47;&#47; check if "can" startScan
    if (shouldCheckCan) {
      &#47;&#47; check if can-startScan
      final can = await WiFiScan.instance.canStartScan();
      &#47;&#47; if can-not, then show error
      if (can != CanStartScan.yes) {
        if (context.mounted) kShowSnackBar(context, "Cannot start scan: $can");
        return;
      }
    }

    &#47;&#47; call startScan API
    final result = await WiFiScan.instance.startScan();
    if (context.mounted) kShowSnackBar(context, "startScan: $result");
    &#47;&#47; reset access points.
    setState(() => accessPoints = []);
  }

  Future _canGetScannedResults(BuildContext context) async {
    if (shouldCheckCan) {
      &#47;&#47; check if can-getScannedResults
      final can = await WiFiScan.instance.canGetScannedResults();
      &#47;&#47; if can-not, then show error
      if (can != CanGetScannedResults.yes) {
        if (context.mounted) {
          kShowSnackBar(context, "Cannot get scanned results: $can");
        }
        accessPoints = [];
        return false;
      }
    }
    return true;
  }

  Future _getScannedResults(BuildContext context) async {
    if (await _canGetScannedResults(context)) {
      &#47;&#47; get scanned results
      final results = await WiFiScan.instance.getScannedResults();
      setState(() => accessPoints = results);
    }
  }

  Future _startListeningToScanResults(BuildContext context) async {
    if (await _canGetScannedResults(context)) {
      subscription = WiFiScan.instance.onScannedResultsAvailable
          .listen((result) => setState(() => accessPoints = result));
    }
  }

  void _stopListeningToScanResults() {
    subscription?.cancel();
    setState(() => subscription = null);
  }

  @override
  void dispose() {
    super.dispose();
    &#47;&#47; stop subscription for scanned results
    _stopListeningToScanResults();
  }
